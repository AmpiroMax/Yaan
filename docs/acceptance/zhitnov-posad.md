## Рецепт (обе руки — ОДИН бинарник, правило 47)

Выпуски — ЧЕРНОВЫЕ, по два прохода каждый (поле высот своё на руку):

    # рука «ДО» — генератор из HEAD на СЕГОДНЯШНЕЙ полке и сегодняшнем паспорте
    git show HEAD:tools/gen_city.py > tools/gen_city_HEADARM.py
    for i in 1 2; do DFN_CITY_HEIGHTS=<был>.txt DFN_CITY_WRITE_HEIGHTS=<был>.txt \
        DFN_GEN_OUT=ch-was python3 tools/gen_city_HEADARM.py cornhall; done
    # рука «ПОСЛЕ»
    for i in 1 2; do DFN_CITY_HEIGHTS=<стал>.txt DFN_CITY_WRITE_HEIGHTS=<стал>.txt \
        DFN_GEN_OUT=ch-now python3 tools/gen_city.py cornhall; done
